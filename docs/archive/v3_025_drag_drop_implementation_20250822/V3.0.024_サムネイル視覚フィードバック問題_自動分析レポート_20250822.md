```csharp
// OSS標準パターン：GiveFeedback イベント処理
private static void OnGiveFeedback(object sender, GiveFeedbackEventArgs e)
{
    if (e.Effects.HasFlag(DragDropEffects.Move))
    {
        Mouse.SetCursor(Cursors.Hand);  // 掴みカーソル
        e.UseDefaultCursors = false;
        e.Handled = true;
    }
    else if (e.Effects.HasFlag(DragDropEffects.Copy))
    {
        Mouse.SetCursor(Cursors.Cross); // コピーカーソル
        e.UseDefaultCursors = false;
        e.Handled = true;
    }
}
```
